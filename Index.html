# College Student Result and Grade System # Using Python print("===================================") print(" COLLEGE STUDENT RESULT SYSTEM") print("===================================") # Enter student details name = input("Enter Student Name: ") roll_no = input("Enter Roll Number: ") # Enter subject marks (out of 100) print("\nEnter Marks for 5 Subjects (out of 100):") sub1 = float(input("Subject 1: ")) sub2 = float(input("Subject 2: ")) sub3 = float(input("Subject 3: ")) sub4 = float(input("Subject 4: ")) sub5 = float(input("Subject 5: ")) marks = [sub1, sub2, sub3, sub4, sub5] # Validate marks if any(mark < 0 or mark > 100 for mark in marks): print("Error: Marks must be between 0 and 100.") else: total = sum(marks) percentage = total / 5 # Check pass or fail if any(mark < 35 for mark in marks): result = "FAIL" grade = "F" else: result = "PASS" if percentage >= 90: grade = "A+" elif percentage >= 80: grade = "A" elif percentage >= 70: grade = "B" elif percentage >= 60: grade = "C" elif percentage >= 50: grade = "D" elif percentage >= 35: grade = "E" # Display student result print("\n===================================") print(" COLLEGE RESULT CARD") print("===================================") print("Student Name :", name) print("Roll Number :", roll_no) print("-----------------------------------") print("Subject 1 :", sub1) print("Subject 2 :", sub2) print("Subject 3 :", sub3) print("Subject 4 :", sub4) print("Subject 5 :", sub5) print("-----------------------------------") print("Total Marks :", total, "/ 500") print("Percentage :", round(percentage, 2), "%") print("Grade :", grade) print("Result :", result) print("===================================") print("Result Generated Successfully!")
